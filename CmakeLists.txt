cmake_minimum_required(VERSION 3.16)
project(chip8)


set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED TRUE)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED TRUE)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

message(STATUS "CMAKE_CXX_COMPILER: ${CMAKE_CXX_COMPILER}")
message(STATUS "CMAKE_CXX_COMPILER_ID: ${CMAKE_CXX_COMPILER_ID}")
message(STATUS "CMAKE_CXX_COMPILER_VERSION: ${CMAKE_CXX_COMPILER_VERSION}")

# set the output directory for built objects.
# This makes sure that the dynamic library goes into the build directory automatically.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIGURATION>")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIGURATION>")

# This assumes the SDL source is available in vendored/SDL
add_subdirectory(vendored/SDL EXCLUDE_FROM_ALL)
add_subdirectory(vendored/SDL_ttf EXCLUDE_FROM_ALL)

# Create your game executable target as usual
# I have no clue how and what to put here, sometimes i need to put something here sometimes I dont so what the fuck
add_executable(Chip8 
    main.cpp 
    res/menu.rc 
    src/chip8/fonts.cpp 
    src/chip8/specs.cpp 
    src/chip8/instructions.cpp 
    src/chip8/input.cpp 
    src/config/menu.cpp
    src/config/database.cpp 
    src/ui/ui.cpp
    res/resource.h
    vendored/ImGui/imgui_impl_sdl3.cpp
    vendored/ImGui/imgui_impl_sdlrenderer3.cpp
    vendored/ImGui/imgui.cpp
    vendored/ImGui/imgui_draw.cpp
    vendored/ImGui/imgui_tables.cpp
    vendored/ImGui/imgui_widgets.cpp
    vendored/ImGui/imgui_demo.cpp
)

# Link to the actual SDL3 library.
target_link_libraries(Chip8 PRIVATE SDL3_ttf::SDL3_ttf SDL3::SDL3 user32 gdi32)

configure_file(res/settings.json ${CMAKE_BINARY_DIR}/Debug/settings.json COPYONLY)
configure_file(res/flags.json ${CMAKE_BINARY_DIR}/Debug/flags.json COPYONLY)

configure_file(res/database/platforms.json ${CMAKE_BINARY_DIR}/Debug/database/platforms.json COPYONLY)
configure_file(res/database/programs.json ${CMAKE_BINARY_DIR}/Debug/database/programs.json COPYONLY)
configure_file(res/database/quirks.json ${CMAKE_BINARY_DIR}/Debug/database/quirks.json COPYONLY)
configure_file(res/database/sha1-hashes.json ${CMAKE_BINARY_DIR}/Debug/database/sha1-hashes.json COPYONLY)
